🔄 Dynamisches Laden

So funktioniert's

Das Widget wird erst bei Bedarf von einer URL geladen. Ideal für Lazy-Loading oder CDN-Integration.

// Widget dynamisch laden
function loadDQMWidget(url, fallbackUrl) {
  const script = document.createElement('script');
  script.src = url;
  script.onerror = () => {
    // Fallback URL versuchen
    const fallback = document.createElement('script');
    fallback.src = fallbackUrl;
    document.head.appendChild(fallback);
  };
  document.head.appendChild(script);
}

// Konfiguration setzen
window.DQM_CONFIG = { apiKey: '...', websiteId: '...' };

// Laden starten
loadDQMWidget(
  'https://cdn.example.com/dqm-widget.iife.js',
  '/dist/dqm-widget.iife.js'
);
